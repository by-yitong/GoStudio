# Values used to calculate the hash in this folder name.
# Should not depend on the absolute path of the project itself.
#   - AGP: 8.13.0.
#   - $NDK is the path to NDK 29.0.14206865.
#   - $PROJECT is the path to the parent folder of the root Gradle build file.
#   - $ABI is the ABI to be built with. The specific value doesn't contribute to the value of the hash.
#   - $HASH is the hash value computed from this text.
#   - $CMAKE is the path to CMake 4.3.0.
#   - $NINJA is the path to Ninja.
-H$PROJECT/modules/terminal-emulator/src/main/cpp
-DCMAKE_SYSTEM_NAME=Android
-DCMAKE_EXPORT_COMPILE_COMMANDS=ON
-DCMAKE_SYSTEM_VERSION=21
-DANDROID_PLATFORM=android-21
-DANDROID_ABI=$ABI
-DCMAKE_ANDROID_ARCH_ABI=$ABI
-DANDROID_NDK=$NDK
-DCMAKE_ANDROID_NDK=$NDK
-DCMAKE_TOOLCHAIN_FILE=$NDK/build/cmake/android.toolchain.cmake
-DCMAKE_MAKE_PROGRAM=$NINJA
-DCMAKE_C_FLAGS=-std=c11 -Wall -Wextra -Werror -Os -fno-stack-protector
-DCMAKE_CXX_FLAGS=-Wall -Wextra -Werror -Os -fno-stack-protector
-DCMAKE_LIBRARY_OUTPUT_DIRECTORY=$PROJECT/modules/terminal-emulator/build/intermediates/cxx/RelWithDebInfo/$HASH/obj/$ABI
-DCMAKE_RUNTIME_OUTPUT_DIRECTORY=$PROJECT/modules/terminal-emulator/build/intermediates/cxx/RelWithDebInfo/$HASH/obj/$ABI
-DCMAKE_BUILD_TYPE=RelWithDebInfo
-B$PROJECT/modules/terminal-emulator/.cxx/RelWithDebInfo/$HASH/$ABI
-GNinja
